# Clean column names first
df.columns = df.columns.str.strip()

# Create Regional Summary
regional_summary = df.groupby("Region").agg(
    COUNT=("Region", "count"),
    Hen_house=("Hen house", "sum"),
    Hen_Day=("Hen Day", "sum"),
    Manure_bags=("No.of Manure( Bags)", "sum"),
    Avg_weight=("No.Kgs at maturity", "mean")
).reset_index()
